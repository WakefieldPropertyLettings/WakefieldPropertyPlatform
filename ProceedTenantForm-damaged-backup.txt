function numericValue(
  value: string | null | undefined
) {
  if (!value) {
    return "";
  }

  /*
   * Handles values such as:
   * 430
   * £430
   * £430.00
   * 430 pcm
   */
  const cleaned = value
    .replace(/,/g, "")
    .match(/\d+(?:\.\d+)?/);

  return cleaned?.[0] ?? "";
}

function calculatePreview(
  rentValue: string,
  moveDateValue: string,
  paymentDayValue: string
) {
  const rent =
    Number(rentValue);

  const paymentDay =
    Number(paymentDayValue);

  if (
    !Number.isFinite(rent) ||
    rent <= 0 ||
    !moveDateValue ||
    ![1, 15].includes(paymentDay)
  ) {
    return null;
  }

  const [
    yearValue,
    monthValue,
    dayValue,
  ] = moveDateValue
    .split("-")
    .map(Number);

  if (
    !yearValue ||
    !monthValue ||
    !dayValue
  ) {
    return null;
  }

  /*
   * Wakefield Property Lettings
   * pro-rata rule:
   *
   * Every rental month is treated as
   * exactly 30 days.
   *
   * Daily rent = monthly rent / 30.
   *
   * Calendar month length (28, 29, 30
   * or 31 days) does NOT change the
   * daily rental rate.
   */
  const dailyRate =
    rent / 30;

  /*
   * If the tenant moves in exactly on
   * their normal rent payment day,
   * there is no pro-rata period.
   *
   * The normal monthly rent is due.
   */
  if (dayValue === paymentDay) {
    return {
      days: 0,
      amount: 0,
      dailyRate:
        Math.round(
          dailyRate * 100
        ) / 100,
      nextPaymentDate:
        moveDateValue,
    };
  }

  let days = 0;

  let nextPaymentYear =
    yearValue;

  let nextPaymentMonth =
    monthValue;

  /*
   * PAYMENT DAY = 1ST
   *
   * Example:
   * Move in 24th September.
   *
   * Rental-month calculation:
   * 24,25,26,27,28,29,30
   * = 7 chargeable days.
   *
   * The next normal monthly rent
   * becomes due on 1st October.
   *
   * A calendar month having 31 days
   * does not create a 31st rental day.
   */
  if (paymentDay === 1) {
    days =
      Math.max(
        0,
        31 - Math.min(
          dayValue,
          30
        )
      );

    nextPaymentMonth += 1;

    if (
      nextPaymentMonth > 12
    ) {
      nextPaymentMonth = 1;
      nextPaymentYear += 1;
    }
  }

  /*
   * PAYMENT DAY = 15TH
   */
  if (paymentDay === 15) {
    if (dayValue < 15) {
      /*
       * Example:
       * move in 10th
       * → charge 10th-14th
       * → normal rent starts 15th.
       */
      days =
        15 - dayValue;
    } else {
      /*
       * Move-in is after the 15th.
       *
       * Treat the rental cycle as
       * 30 days rather than using
       * the real calendar length.
       *
       * Example:
       * 24th → next 15th
       * 7 days to complete the
       * 30-day month + 14 days
       * before the next 15th.
       */
      days =
        (31 -
          Math.min(
            dayValue,
            30
          )) +
        14;

      nextPaymentMonth += 1;

      if (
        nextPaymentMonth > 12
      ) {
        nextPaymentMonth = 1;
        nextPaymentYear += 1;
      }
    }
  }

  /*
   * Never allow a pro-rata period to
   * become a full month or more.
   *
   * A full rental month is simply the
   * monthly rent.
   */
  days =
    Math.max(
      0,
      Math.min(days, 29)
    );

  const amount =
    Math.round(
      dailyRate *
        days *
        100
    ) / 100;

  const nextPaymentDate =
    `${nextPaymentYear}-${String(
      nextPaymentMonth
    ).padStart(
      2,
      "0"
    )}-${String(
      paymentDay
    ).padStart(
      2,
      "0"
    )}`;

  return {
    days,
    amount,
    dailyRate:
      Math.round(
        dailyRate * 100
      ) / 100,
    nextPaymentDate,
  };
}
